  <script>
    // Mapeamento das rotas da API do ViewDNS.info
    const apiEndpoints = {
      reversewhois: { endpoint: 'reversewhois', param: 'q', label: 'E-mail ou Termo de Busca:' },
      whois: { endpoint: 'whois', param: 'domain', label: 'Domínio ou IP:' },
      dns: { endpoint: 'dns', param: 'domain', label: 'Domínio:' },
      iphistory: { endpoint: 'iphistory', param: 'domain', label: 'Domínio:' },
      reverseip: { endpoint: 'reverseip', param: 'host', label: 'Endereço IP ou Domínio:' },
      iplocation: { endpoint: 'iplocation', param: 'ip', label: 'Endereço IP (IPv4):' },
      ping: { endpoint: 'ping', param: 'host', label: 'Host ou Domínio:' },
      portscan: { endpoint: 'portscan', param: 'host', label: 'Host ou Domínio:' },
      spamdblookup: { endpoint: 'spamdblookup', param: 'ip', label: 'Endereço IP:' },
      abusecontact: { endpoint: 'abusecontact', param: 'domain', label: 'Domínio:' }
    };

    function atualizarFormulario() {
      const selectedTool = document.getElementById('toolSelect').value;
      const config = apiEndpoints[selectedTool];
      document.getElementById('inputLabel').textContent = config.label;
    }

    async function executarAPI() {
      const apiKey = document.getElementById('apiKey').value.trim();
      const toolKey = document.getElementById('toolSelect').value;
      const queryValue = document.getElementById('queryInput').value.trim();
      const container = document.getElementById('resultadoContainer');
      const loading = document.getElementById('loading');

      if (!apiKey) {
        alert('Por favor, insira sua chave de API do ViewDNS.');
        return;
      }
      if (!queryValue) {
        alert('Por favor, preencha o campo de consulta.');
        return;
      }

      container.innerHTML = '';
      loading.style.display = 'block';

      const config = apiEndpoints[toolKey];
      const url = `https://api.viewdns.info/${config.endpoint}/?${config.param}=${encodeURIComponent(queryValue)}&apikey=${apiKey}&output=json`;

      try {
        const response = await fetch(url);
        const data = await response.json();

        loading.style.display = 'none';

        // Renderiza tabela se houver matches (ex: reversewhois, dns, etc) ou JSON formatado
        if (data.response && data.response.matches && Array.isArray(data.response.matches)) {
          let html = `<table><thead><tr>`;
          const keys = Object.keys(data.response.matches[0]);
          keys.forEach(k => html += `<th>${k}</th>`);
          html += `</tr></thead><tbody>`;

          data.response.matches.forEach(item => {
            html += `<tr>`;
            keys.forEach(k => html += `<td>${item[k] || ''}</td>`);
            html += `</tr>`;
          });
          html += `</tbody></table>`;
          container.innerHTML = html;
        } else {
          // Exibe o JSON estruturado puro para as demais respostas
          container.innerHTML = `<div class="result-area">${JSON.stringify(data, null, 2)}</div>`;
        }

      } catch (error) {
        loading.style.display = 'none';
        container.innerHTML = `<div class="result-area" style="color: #ff6b6b;">Erro de conexão ou resposta inválida: ${error.message}</div>`;
      }
    }
  </script>
