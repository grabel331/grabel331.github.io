  <script>
    const navBar = document.getElementById("navBar");
    
    navBar.addEventListener("click", (e) => {
      if (!e.target.classList.contains("nav-btn")) return;

      // Atualiza o botão ativo
      document.querySelectorAll(".nav-btn").forEach(btn => btn.classList.remove("active"));
      e.target.classList.add("active");

      const targetName = e.target.dataset.target;

      // Esconde todas as visões
      document.getElementById("view-worldometer").classList.remove("active");
      document.getElementById("view-brazil").classList.remove("active");

      // Mostra a visão selecionada
      if (targetName === "worldometer") {
        document.getElementById("view-worldometer").classList.add("active");
      } else if (targetName === "brazil") {
        document.getElementById("view-brazil").classList.add("active");
      }
    });
  </script>
